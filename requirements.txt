from __future__ import annotations

import os
import textwrap
import uuid
from pathlib import Path
from typing import Dict

from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel
from gtts import gTTS
from moviepy.editor import AudioFileClip, ImageClip, concatenate_videoclips
from PIL import Image, ImageDraw, ImageFont

BASE_DIR = Path(__file__).resolve().parent
STATIC_DIR = BASE_DIR / "static"
TEMPLATES_DIR = BASE_DIR / "templates"
OUTPUT_DIR = BASE_DIR / "generated_videos"
OUTPUT_DIR.mkdir(exist_ok=True)

app = FastAPI(title="Text to Video Generator")
app.mount("/static", StaticFiles(directory=str(STATIC_DIR)), name="static")


class VideoRequest(BaseModel):
    text: str
    voice: str = "en"
    user_id: str | None = None


class PurchaseRequest(BaseModel):
    user_id: str | None = None


# Simple in-memory user credit system.
# This can be swapped with SQLite or PostgreSQL later.
USER_CREDITS: Dict[str, int] = {}
FREE_VIDEO_LIMIT = 5
PURCHASE_BONUS = 10


def get_user_id(user_id: str | None) -> str:
    if user_id and user_id.strip():
        return user_id.strip()
    return "guest-user"


def get_user_credits(user_id: str) -> int:
    return USER_CREDITS.get(user_id, FREE_VIDEO_LIMIT)


def set_user_credits(user_id: str, amount: int) -> int:
    USER_CREDITS[user_id] = max(0, amount)
    return USER_CREDITS[user_id]


def consume_credit(user_id: str) -> bool:
    user_id = get_user_id(user_id)
    current = get_user_credits(user_id)
    if current <= 0:
        return False
    set_user_credits(user_id, current - 1)
    return True


def purchase_credits(user_id: str | None) -> int:
    user_id = get_user_id(user_id)
    next_total = get_user_credits(user_id) + PURCHASE_BONUS
    return set_user_credits(user_id, next_total)


def split_text_into_chunks(text: str, max_chars: int = 120) -> list[str]:
    normalized = " ".join(text.split())
    sentences = [part.strip() for part in normalized.split(".") if part.strip()]
    chunks: list[str] = []
    current = ""

    for sentence in sentences:
        if len(current) + len(sentence) + 1 <= max_chars:
            current = f"{current} {sentence}".strip()
        else:
            if current:
                chunks.append(current)
            current = sentence

    if current:
        chunks.append(current)

    if not chunks:
        words = normalized.split()
        for i in range(0, len(words), 24):
            chunks.append(" ".join(words[i : i + 24]))

    return chunks


def resolve_font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans.ttf", size)
    except Exception:
        return ImageFont.load_default()


def create_scene_image(text: str, output_path: str, index: int) -> None:
    width, height = 1280, 720
    image = Image.new("RGB", (width, height), "#0f172a")
    draw = ImageDraw.Draw(image)

    for y in range(height):
        r = int(15 + (y / height) * 35)
        g = int(23 + (y / height) * 60)
        b = int(42 + (y / height) * 70)
        draw.line((0, y, width, y), fill=(r, g, b))

    title = "Text to Video"
    title_font = resolve_font(30)
    subtitle_font = resolve_font(24)
    body_font = resolve_font(48)

    title_box = draw.textbbox((0, 0), title, font=title_font)
    title_x = (width - (title_box[2] - title_box[0])) / 2
    draw.text((title_x, 60), title, font=title_font, fill=(255, 255, 255))

    card_x1, card_y1 = 120, 170
    card_x2, card_y2 = width - 120, height - 120
    draw.rounded_rectangle(
        (card_x1, card_y1, card_x2, card_y2),
        radius=28,
        fill=(15, 23, 42, 180),
        outline=(71, 85, 105),
        width=2,
    )

    wrapped = textwrap.wrap(text, width=30)
    lines = wrapped[:5]
    y_cursor = 240
    for line in lines:
        line_box = draw.textbbox((0, 0), line, font=body_font)
        line_width = line_box[2] - line_box[0]
        x = (width - line_width) / 2
        draw.text((x, y_cursor), line, font=body_font, fill=(248, 250, 252))
        y_cursor += 70

    page_text = f"Scene {index + 1}"
    page_box = draw.textbbox((0, 0), page_text, font=subtitle_font)
    page_x = width - (page_box[2] - page_box[0]) - 140
    draw.text((page_x, 610), page_text, font=subtitle_font, fill=(148, 163, 184))

    image.save(output_path)


def generate_video_from_text(text: str, voice: str) -> str:
    if not text or not text.strip():
        raise ValueError("Text input cannot be empty.")

    chunks = split_text_into_chunks(text)
    scene_dir = OUTPUT_DIR / str(uuid.uuid4())
    scene_dir.mkdir(exist_ok=True)

    for i, chunk in enumerate(chunks):
        image_path = scene_dir / f"scene_{i}.png"
        create_scene_image(chunk, str(image_path), i)

    audio_path = scene_dir / "voice.mp3"
    tts = gTTS(text=text, lang=voice, slow=False)
    tts.save(str(audio_path))

    clips = []
    for i, chunk in enumerate(chunks):
        image_path = scene_dir / f"scene_{i}.png"
        scene_duration = max(2.5, min(5.0, len(chunk.split()) * 0.25))
        clip = ImageClip(str(image_path), duration=scene_duration)
        clips.append(clip)

    final_video_path = scene_dir / "output_video.mp4"
    video = concatenate_videoclips(clips, method="compose")
    audio = AudioFileClip(str(audio_path))

    if audio.duration > video.duration:
        video = video.set_duration(audio.duration)

    video = video.set_audio(audio)
    video.write_videofile(
        str(final_video_path),
        fps=24,
        codec="libx264",
        audio_codec="aac",
        temp_audiofile=str(scene_dir / "temp_audio.m4a"),
        remove_temp=True,
    )

    return str(final_video_path)


@app.get("/")
def home():
    return FileResponse(str(TEMPLATES_DIR / "index.html"))


@app.get("/api/credits/{user_id}")
def get_credits(user_id: str):
    user_id = get_user_id(user_id)
    remaining = get_user_credits(user_id)
    return {
        "user_id": user_id,
        "remaining_credits": remaining,
        "free_limit": FREE_VIDEO_LIMIT,
        "purchase_bonus": PURCHASE_BONUS,
        "price_usd": 2,
    }


@app.post("/api/generate")
def generate_video(request: VideoRequest):
    user_id = get_user_id(request.user_id)
    remaining = get_user_credits(user_id)

    if remaining <= 0:
        return {
            "success": False,
            "message": "You have used all free videos. Pay $2 for 10 more videos.",
            "need_purchase": True,
            "price_usd": 2,
            "credits_added": PURCHASE_BONUS,
            "remaining_credits": 0,
        }

    try:
        output_path = generate_video_from_text(request.text, request.voice)
        consume_credit(user_id)
        filename = output_path.split("/")[-1]
        return {
            "success": True,
            "message": "Video generated successfully.",
            "video_url": f"/video/{filename}",
            "filename": filename,
            "remaining_credits": get_user_credits(user_id),
        }
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc)) from exc


@app.post("/api/purchase")
def purchase(request: PurchaseRequest):
    user_id = get_user_id(request.user_id)
    updated = purchase_credits(user_id)
    return {
        "success": True,
        "message": "Purchase successful. You received 10 extra videos.",
        "price_usd": 2,
        "credits_added": PURCHASE_BONUS,
        "remaining_credits": updated,
    }


@app.get("/video/{filename}")
def get_video(filename: str):
    for candidate in OUTPUT_DIR.rglob(filename):
        if candidate.is_file():
            return FileResponse(str(candidate), media_type="video/mp4")
    raise HTTPException(status_code=404, detail="Video not found.")


if __name__ == "__main__":
    import uvicorn

    uvicorn.run(app, host="0.0.0.0", port=8000)
